import hashlib,subprocess,time
from pathlib import Path
from tools.platform_host import Host
from tools.platform_store import plain
from tools.research import core
from tools.research.state import state,save,update
from tools.state_io import read,atomic_json
from extensions.tendon_family.sorosim_codesign import integrated_control_source
root=Path.cwd();started=time.perf_counter();name='sorosim-s56-codesign-20261017';h=Host(root/'runs/research'/name,name)
index=read(root/'evidence/sorosim_s56/codesign/summary.json');before=state(h)

assert hashlib.sha256(body).hexdigest()==reference['sha256']
with h.store.transaction() as db:blob=plain(h.store.put(db,body,media='application/octet-stream'))
ref=save(h,dict(contract=reference,raw_bytes=blob,source_run=residual['run_id'],purpose='Preserve the exact compatible residual reference; no change to checkpoint/configuration or physical success classification'),'residual_reference_preservation')
fixture=save(h,read(root/'runs/s56_grid_boundary_check.json'),'interface_arithmetic_fixture_only')

    original_task_satisfied=False,numerical_integration_complete=True,
    reason='Legacy handoff used historical verified-trajectory naming and a completeness flag while original task failed; it also copied coarse slacks/objective into high-resolution sampled states. New kind separates numerical completeness and original-task verification and omits coarse epigraph fields.',
    original_records_preserved=True,actual_controls_and_mathematical_outputs_identical=True,
    impact='Actual P3 playback consumed only unchanged physical commands; P5 initialization consumed only controls/initial state and rematerialized its own nodes. No original task evaluator or numerical outcome changed; no experiment repeated.',
    interface_fix=dict(reason='round the already grid-validated remaining interval count before lower-bound comparison',validation=fixture,physical_reexecution=False),
    residual_reference=ref,feedback=packet,postprocessing_commit=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),
    scientific_cost=dict(full_nlp=0,physical_steps=0,ppo_actions=0,provider_requests=0))

    erratum=err,residual_reference=ref,interface_fixture=fixture,agent_feedback=packet,
    original_stage_source='evidence/sorosim_s56/codesign/study',append_only=True,
    scientific_counters_unchanged=True,postprocessing_elapsed_s=time.perf_counter()-started)
atomic_json(root/'evidence/sorosim_s56/annotations/summary.json',summary)
print(summary)
