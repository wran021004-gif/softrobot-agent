provider={};actions=[];last_packet=None;last_send=None;reads=[];retrievals=[];errors=[]
for event in events:
    kind=event['kind']
    if kind=='model_decision_packet':last_packet=event['outputs'][0]
    elif kind=='r3_request':last_send=event['request_id']
    elif kind=='r3_response' and event['status']=='metadata_saved':
        provider[event['request_id']]=dict(request_id=event['request_id'],**host.store.artifact(event['outputs'][0]))
    elif kind=='model_original_tool_request':
        use=host.store.artifact(event['outputs'][0])['tool_use']
        actions.append(dict(tool_use=use,provider_request_id=last_send,decision_packet=last_packet,
            prior_successful_public_reads=deepcopy(reads),prior_successful_context_retrievals=deepcopy(retrievals),
            ownership='Agent-selected original native request'))
    elif kind=='model_original_tool_feedback':
        value=host.store.artifact(event['outputs'][0]);result=value['result'];use=value['tool_use']
        row=next(a for a in reversed(actions) if a['tool_use']['toolUseId']==use['toolUseId'])
        row['feedback_reference']=event['outputs'][0];row['feedback_status']=result['status']
        if result['status']=='success' and use['name']=='evidence_read':reads.append(use['input'])
        if result['status']=='success' and use['name']=='retrieve_context':retrievals.append(use['input'])
        if result['status']=='error':
            body=json.loads(result['content'][0]['text'])
            errors.append(dict(tool=use['name'],operation=use['input'].get('action') or use['input'].get('operation'),
                message=body.get('message'),request=use['toolUseId'],provider_request_id=last_send))
assert len(provider)==26 and all(row['http_status']==200 for row in provider.values())

# Verify original archive bytes/member hashes and inspect actual credential matches
# without printing, hashing for publication, or saving the credential itself.
load_credential(Path(s['spec']['runtime']['credentials_reference']).expanduser())
import os
secret=os.environ['DEEPSEEK_API_KEY'].encode();archives=[]
for folder in (DEST/'study',DEST/'checkpoints/study'):
    manifest=read(folder/'archive_manifest.json');path=folder/'immutable_artifacts.tar.gz'
    assert hashlib.file_digest(path.open('rb'),'sha256').hexdigest()==manifest['archive_sha256']
    expected={r['name']:r['sha256'] for r in manifest['members']};members=0
    with tarfile.open(path) as archive:
        for item in archive:
            if not item.isfile():continue
            body=archive.extractfile(item).read();assert hashlib.sha256(body).hexdigest()==expected[item.name]
            assert secret not in body,'Credential found in artifact'
            if body.startswith(b'PK'):
                with zipfile.ZipFile(BytesIO(body)) as nested:
                    assert all(secret not in nested.read(name) for name in nested.namelist()),'Credential found in checkpoint'
            members+=1
    assert members==len(expected)
    archives.append(dict(path=str(path.relative_to(ROOT)),sha256=manifest['archive_sha256'],members=members,bytes=path.stat().st_size,member_hashes_verified=True,credential_matches=0))
assert read(DEST/'checkpoints/summary.json')['primary_archive_sha256']==archives[0]['sha256']
