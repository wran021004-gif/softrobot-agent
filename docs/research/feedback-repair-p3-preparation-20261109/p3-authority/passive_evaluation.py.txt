"""Read-only, configuration-driven passive load response metrics; no SELECT claim."""
import numpy as np


def response_metrics(forces, displacements, uncertainty, thresholds):
    f=np.asarray(forces);d=np.asarray(displacements);increments=np.diff(d)
    precision=max(thresholds['denominator_min_m'],thresholds['denominator_uncertainty_multiplier']*uncertainty)
    valid=bool(np.all(np.isfinite(d)) and np.all(np.isfinite(f)) and np.all(np.diff(f)>0) and np.all(increments>precision))
    k=(np.diff(f)/increments).tolist() if valid else None
    return dict(signed_displacements_m=d.tolist(),secant_stiffness_n_m=[float(x/y) if y>precision else None for x,y in zip(f,d)],
        interval_stiffness_n_m=k,stiffness_ratio=k[-1]/k[0] if valid else None,
        denominator_status='valid' if valid else 'nonpositive_or_below_precision',denominator_precision_m=precision)


def evaluate(protocol, raw):
    p=protocol;a=p['acceptance'];samples=raw['samples'];tails=[];checks={};stages=[]
    if raw['status']!='completed':return dict(status='INVALID',reason=raw.get('invalid'),metrics=None)
    end=0.
    for stage in p['stages']:
        end+=stage['duration_s']
        rows=[r for r in samples if r['stage']==stage['name'] and r['time_s']>end-p['tail_s']-1e-8]
        if len(rows)<round(p['tail_s']/p['scoring_interval_s']):return dict(status='UNKNOWN',reason='incomplete_tail',metrics=None)
        positions=np.array([r['monitor_positions'] for r in rows]);point=np.array([r['point'] for r in rows])
        drift=float(np.max(np.linalg.norm(positions-positions[0],axis=2)))
        local=dict(point_speed=max(np.linalg.norm(r['velocity']) for r in rows)<=p['monitor_speed_limit_m_s'],
            monitor_speed=max(r['monitor_speed'] for r in rows)<=p['monitor_speed_limit_m_s'],
            qvel=max(max(abs(np.array(r['qvel']))) for r in rows)<=p['qvel_inf_limit_rad_s'],
            tail_drift=drift<=p['tail_drift_limit_m'],static_residual=max(r['static_balance'] for r in rows)<=p['static_balance_inf_limit_nm'])
        tails.append(rows);stages.append(dict(name=stage['name'],checks=local,stable=all(local.values()),drift_m=drift,
            mean_point_m=point.mean(axis=0).tolist(),point_spread_m=float(np.max(np.linalg.norm(point-point.mean(axis=0),axis=1))),
            max_static_residual_nm=max(r['static_balance'] for r in rows),max_qvel=max(max(abs(np.array(r['qvel']))) for r in rows)))
    e=np.array(raw['direction']);p0=np.array(stages[0]['mean_point_m'])
    loaded=[i for i,stage in enumerate(p['stages']) if stage['force_n']>0]
    displacement=[float((np.array(stages[i]['mean_point_m'])-p0)@e) for i in loaded]
    uncertainty=max(stage['point_spread_m'] for stage in stages)
    metrics=response_metrics([p['stages'][i]['force_n'] for i in loaded],displacement,uncertainty,a)
    residual=float(np.linalg.norm(np.array(stages[-1]['mean_point_m'])-p0));metrics['residual_m']=residual
    low=np.array([r['tension'] for r in tails[loaded[0]]]);high=np.array([r['tension'] for r in tails[loaded[-1]]])
    slack=np.array([r['length_residual'] for r in tails[loaded[0]]]);upper=np.array([r['length_residual'] for r in tails[loaded[-1]]])
    activated=[]
    for t in range(low.shape[1]):
        active=bool(max(low[:,t])<=a['low_tension_max_n'] and min(high[:,t])>=a['high_tension_min_n']
            and max(slack[:,t])<=-a['low_slack_min_m'] and max(abs(upper[:,t]))<=a['high_constraint_residual_max_m']
            and all(r['constraint_state'][t]>0 and r['constraint_residual'][t] is not None for r in tails[loaded[-1]]))
        activated.append(dict(tendon=t,activated=active,low_max_n=float(max(low[:,t])),high_min_n=float(min(high[:,t])),
            low_max_length_residual_m=float(max(slack[:,t])),high_max_length_residual_m=float(max(upper[:,t]))))
    checks.update(compliant=displacement[0]>=a['low_displacement_min_m'],
        ordered_and_bounded=0<displacement[0]<displacement[1]<displacement[2]<=a['high_displacement_max_m'],
        progressive=metrics['stiffness_ratio'] is not None and metrics['stiffness_ratio']>=a['stiffness_ratio_min'],
        recovered=residual<=a['residual_max_m'],stable=all(s['stable'] for s in stages),
        tendon_switch=any(t['activated'] for t in activated),route_valid=raw.get('geometry',{}).get('status')=='legal_at_checked_poses')
    if not checks['stable']:status='INVALID'
    elif not checks['route_valid']:status='INVALID' if raw.get('geometry',{}).get('status')=='illegal' else 'UNKNOWN'
    elif metrics['denominator_status']!='valid':status='UNKNOWN'
    else:status='PASS' if all(checks.values()) else 'FAIL'
    return dict(status=status,checks=checks,metrics=metrics,stages=stages,tendon_transitions=activated,
        scope='development evaluator; '+raw.get('stability_grid','score grid')+' stability; full-step constraints in raw; route pose scope in geometry')


def mechanism(protocol, constrained, unrestricted):
    """Only valid, stable paired observations can support a causal comparison."""
    if any(r['status'] in ('INVALID','UNKNOWN') or not r.get('checks',{}).get('stable') for r in (constrained,unrestricted)):
        return dict(status='UNKNOWN',reason='invalid_or_unsettled_pair')
    c=constrained['metrics'];u=unrestricted['metrics']
    increase=u['signed_displacements_m'][-1]/c['signed_displacements_m'][-1]
    ratio=u['interval_stiffness_n_m'][-1]/c['interval_stiffness_n_m'][-1]
    passed=increase>=protocol['mechanism']['high_displacement_increase_min'] or ratio<=protocol['mechanism']['late_stiffness_ratio_max']
    return dict(status='PASS' if passed else 'FAIL',high_displacement_ratio=increase,late_stiffness_ratio=ratio)
