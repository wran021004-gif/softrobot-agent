"""Continuous world-vector load schedules on existing passive native models.

Records a refreshed post-step state with the force actually held over that step.
No state is reset at stage boundaries. Legacy PassiveModel.run is unchanged.
"""
import csv
from pathlib import Path
import numpy as np
import mujoco
from .passive import material_point


def apply_world_force(native, data, binding, force):
    data.qfrc_applied[:]=0.;data.xfrc_applied[:]=0.
    position,jacobian,body=native.point(data,binding)
    mujoco.mj_applyFT(native.model,data,np.asarray(force,dtype=float),np.zeros(3),
                     position,body,data.qfrc_applied)
    return position,jacobian


def constraint_observation(native,data):
    tension=np.zeros(native.model.ntendon);state=np.full(native.model.ntendon,-1.)
    residual=np.full(native.model.ntendon,np.nan);direction_error=0.
    J=native.tendon_jac(data)
    E=np.zeros((data.nefc,native.model.nv))
    if mujoco.mj_isSparse(native.model):
        mujoco.mju_sparse2dense(E,data.efc_J,data.efc_J_rownnz,data.efc_J_rowadr,data.efc_J_colind)
    else:E[:]=data.efc_J.reshape(data.nefc,native.model.nv)
    for r in range(data.nefc):
        if data.efc_type[r]!=int(mujoco.mjtConstraint.mjCNSTR_LIMIT_TENDON):continue
        t=int(data.efc_id[r]);tension[t]+=data.efc_force[r]
        state[t]=data.efc_state[r];residual[t]=data.efc_pos[r]
        direction_error=max(direction_error,float(np.max(abs(E[r]+J[t]))))
    return tension,state,residual,direction_error


def run_schedule(native, protocol, direction, initial, path, attempt):
    """Full-step CSV plus separate score-grid observations; counts before mj_step."""
    m=native.model;p=protocol;dt=float(m.opt.timestep);d=mujoco.MjData(m)
    direction=np.asarray(direction,dtype=float)
    if direction.shape!=(3,) or abs(np.linalg.norm(direction)-1)>1e-10:raise ValueError('UNIT_DIRECTION_REQUIRED')
    binding=material_point(native.physics,p['force_point']['s'])
    d.qpos[:]=initial;mujoco.mj_forward(m,d)
    columns=['time_s','stage','force_x_n','force_y_n','force_z_n']
    columns += [f'qpos_{i}' for i in range(m.nq)]+[f'qvel_{i}' for i in range(m.nv)]
    columns += ['point_x_m','point_y_m','point_z_m','velocity_x_m_s','velocity_y_m_s','velocity_z_m_s']
    for name in ('length_m','tension_n','constraint_state','constraint_residual_m','length_residual_m'):
        columns += [f'{name}_{i}' for i in range(m.ntendon)]
    columns += ['contacts','warnings','direction_error','force_map_error']
    samples=[];steps=0;previous_force=np.zeros(3);invalid=[];maxima=dict(excess=0.,capacity=0.,direction=0.,force_map=0.,contacts=0,warnings=0)
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True)
    every=round(p['scoring_interval_s']/dt)
    with path.open('x',newline='') as stream:
        writer=csv.writer(stream);writer.writerow(columns)
        for stage_index,stage in enumerate(p['stages']):
            count=round(stage['duration_s']/dt);target=direction*stage['force_n'];start=previous_force.copy()
            for k in range(count):
                # Endpoint-held smoothstep: first integration sees a small positive ramp value.
                u=min((k+1)*dt/p['ramp_s'],1.);a=u*u*(3-2*u);force=start+(target-start)*a
                _,applied_J=apply_world_force(native,d,binding,force)
                map_error=float(np.max(abs(d.qfrc_applied-applied_J.T@force)))
                old=float(d.time);attempt();steps+=1;mujoco.mj_step(m,d)
                if not np.isfinite(d.qpos).all() or not np.isfinite(d.qvel).all() or d.time<=old:
                    invalid.append('numerical_failure');break
                mujoco.mj_forward(m,d)
                point,J,_=native.point(d,binding);velocity=J@d.qvel
                tension,state,residual,direction_error=constraint_observation(native,d)
                excess=d.ten_length-native.payout
                capacities=np.array([t['force_limit_n'] for t in native.physics['tendons']])
                maxima['excess']=max(maxima['excess'],float(max(excess)))
                maxima['capacity']=max(maxima['capacity'],float(max(tension/capacities)))
                maxima['direction']=max(maxima['direction'],direction_error)
                maxima['force_map']=max(maxima['force_map'],map_error)
                maxima['contacts']=max(maxima['contacts'],int(d.ncon))
                maxima['warnings']=max(maxima['warnings'],int(sum(d.warning.number)))
                writer.writerow(np.r_[d.time,stage_index,force,d.qpos,d.qvel,point,velocity,
                    d.ten_length,tension,state,residual,excess,d.ncon,sum(d.warning.number),direction_error,map_error])
                if (k+1)%every==0 or k==count-1:
                    static=mujoco.MjData(m);static.qpos[:]=d.qpos
                    mujoco.mj_forward(m,static);apply_world_force(native,static,binding,force);mujoco.mj_forward(m,static)
                    balance=static.qfrc_passive+static.qfrc_applied-static.qfrc_bias+static.qfrc_constraint
                    monitored=[native.point(d,b) for b in native.points]
                    samples.append(dict(time_s=float(d.time),stage=stage['name'],force=force.tolist(),
                        point=point.tolist(),velocity=velocity.tolist(),qpos=d.qpos.tolist(),qvel=d.qvel.tolist(),
                        monitor_positions=[v[0].tolist() for v in monitored],
                        monitor_speed=max(float(np.linalg.norm(v[1]@d.qvel)) for v in monitored),
                        static_balance=float(max(abs(balance))),length_residual=excess.tolist(),
                        tension=tension.tolist(),constraint_state=state.tolist(),
                        constraint_residual=[None if np.isnan(x) else float(x) for x in residual]))
                if d.ncon or np.any(d.warning.number) or min(tension)<-1e-10 or direction_error>1e-8 or map_error>1e-10:
                    invalid.append('contact_warning_or_constraint_direction');break
                if not native.unrestricted and (max(excess)>p['length_excess_limit_m'] or max(tension/capacities)>1.):
                    invalid.append('tendon_excess_or_capacity');break
            previous_force=target
            if invalid:break
    return dict(status='INVALID' if invalid else 'completed',invalid=invalid,samples=samples,steps=steps,
                timestep_s=dt,columns=columns,trajectory=str(path),full_step_maxima=maxima,
                direction=direction.tolist(),unrestricted=native.unrestricted)


def read_schedule(native,protocol,direction,path,*,full_tail=False):
    """Recompute score observations from read-only full-step CSV in a fresh process.

    Every stored state is checked for timestep/stage/force coverage and constraint
    validity. Static residuals are freshly recomputed at the separate score grid.
    Reading saved states is not a new integration or dynamic replication.
    """
    m=native.model;p=protocol;dt=float(m.opt.timestep);d=mujoco.MjData(m)
    path=Path(path)
    with path.open() as f:columns=next(csv.reader(f))
    values=np.loadtxt(path,delimiter=',',skiprows=1,ndmin=2);col={name:i for i,name in enumerate(columns)}
    get=lambda name:values[:,col[name]]
    block=lambda prefix,n:values[:,[col[f'{prefix}_{i}'] for i in range(n)]]
    q=block('qpos',m.nq);v=block('qvel',m.nv);tension=block('tension_n',m.ntendon);excess=block('length_residual_m',m.ntendon)
    state=block('constraint_state',m.ntendon);residual=block('constraint_residual_m',m.ntendon)
    force=values[:,[col[f'force_{axis}_n'] for axis in 'xyz']];time=get('time_s');stage=get('stage').astype(int)
    finite=np.isfinite(np.c_[q,v,tension,excess,force,time]).all();invalid=[]
    counts=[round(s['duration_s']/dt) for s in p['stages']];expected=sum(counts)
    if len(values)!=expected or not finite or max(abs(time-np.arange(1,len(values)+1)*dt))>1e-7:invalid.append('coverage_or_nonfinite')
    expected_stage=np.repeat(np.arange(len(counts)),counts)
    if len(values)==expected and not np.array_equal(stage,expected_stage):invalid.append('stage_coverage')
    expected_force=[];previous=0.
    for s,count in zip(p['stages'],counts):
        u=np.minimum(np.arange(1,count+1)*dt/p['ramp_s'],1.);a=u*u*(3-2*u)
        expected_force.extend((previous+(s['force_n']-previous)*a)[:,None]*np.asarray(direction))
        previous=s['force_n']
    if len(values)==expected and np.max(abs(force-np.array(expected_force)))>1e-10:invalid.append('force_schedule')
    capacity=np.array([t['force_limit_n'] for t in native.physics['tendons']])
    maxima=dict(excess=float(np.max(excess)),capacity=float(np.max(tension/capacity)),direction=float(max(get('direction_error'))),
        force_map=float(max(get('force_map_error'))),contacts=int(max(get('contacts'))),warnings=int(max(get('warnings'))))
    if (maxima['contacts'] or maxima['warnings'] or maxima['direction']>1e-8 or maxima['force_map']>1e-10
            or np.min(tension)<-1e-10 or maxima['capacity']>1.):invalid.append('full_step_physical_validity')
    if not native.unrestricted and maxima['excess']>p['length_excess_limit_m']:invalid.append('tendon_excess')
    binding=material_point(native.physics,p['force_point']['s']);samples=[];every=round(p['scoring_interval_s']/dt)
    indices=set(range(every-1,len(values),every))
    if full_tail:
        end=0
        for count in counts:
            end+=count
            indices.update(range(max(0,end-round(p['tail_s']/dt)),min(end,len(values))))
    for i in sorted(indices):
        d.qpos[:]=q[i];d.qvel[:]=v[i];mujoco.mj_forward(m,d)
        point,J,_=native.point(d,binding);items=[native.point(d,b) for b in native.points]
        static=mujoco.MjData(m);static.qpos[:]=q[i];mujoco.mj_forward(m,static)
        apply_world_force(native,static,binding,force[i]);mujoco.mj_forward(m,static)
        balance=static.qfrc_passive+static.qfrc_applied-static.qfrc_bias+static.qfrc_constraint
        # Saved kinematics must agree with the XML, independently of score fields.
        stored_point=values[i,[col[f'point_{axis}_m'] for axis in 'xyz']]
        if np.max(abs(point-stored_point))>1e-10:invalid.append('point_kinematics')
        samples.append(dict(time_s=float(time[i]),stage=p['stages'][stage[i]]['name'],force=force[i].tolist(),
            point=point.tolist(),velocity=(J@v[i]).tolist(),qpos=q[i].tolist(),qvel=v[i].tolist(),
            monitor_positions=[v[0].tolist() for v in items],monitor_speed=max(float(np.linalg.norm(b[1]@v[i])) for b in items),
            static_balance=float(max(abs(balance))),length_residual=excess[i].tolist(),tension=tension[i].tolist(),
            constraint_state=state[i].tolist(),constraint_residual=[None if np.isnan(x) else float(x) for x in residual[i]]))
    return dict(status='INVALID' if invalid else 'completed',invalid=sorted(set(invalid)),samples=samples,steps=len(values),
        timestep_s=dt,trajectory=str(path),columns=columns,direction=list(direction),full_step_maxima=maxima,
        unrestricted=native.unrestricted,provenance='CSV reanalysis; no integration',
        stability_grid='every physical step in each scoring tail' if full_tail else 'score grid')
